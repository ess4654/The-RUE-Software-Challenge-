% ---- THE TWO SWITCHES -------------------------------------------------
% \ufocompacttrue   widen the text block (outer margins pulled outward:
%                   right margin of a recto further right, left margin of a
%                   verso further left) so the same words occupy fewer pages.
%                   \ufocompactfalse restores the original roomy block.
% \ufoprinttrue     KDP PRINT layout: margins are MIRRORED, with the extra

% ---- wrapfig parshape scoping fix ---------------------------------------
% wrapfig steers the text around a figure by installing \parshape from inside
% \everypar. \parshape is a LOCAL assignment, and a paragraph that opens with
% a group -- \textbf{...}, \emph{...}, \hi{...}, or a bare brace, which is how
% every equipment entry, roster biography and highlighted lead sentence in this
% book begins -- starts its first character INSIDE that group. \everypar then
% fires inside the group too, so the \parshape is discarded at the closing
% brace and TeX breaks the paragraph with no shape at all: the text is set at
% the full measure straight across the picture, its caption and its credit line.
% This is the single cause of both the "no wrap" and the "overprint" faults.
% The cure is to make wrapfig's own \parshape assignments global, so a group
% boundary cannot throw the shape away. \WF@finale (which ends the wrapping)
% is made global to match, so nothing can leak past the figure either.
\makeatletter
\def\WF@wraphand{%
\ifnum\c@WF@wrappedlines<\tw@ \WF@finale
\else \begingroup
 \@tempcnta\@ne \let\WF@wrapil\relax \gdef\WF@ps{}%
 \@whilenum \@tempcnta<\c@WF@wrappedlines\do{%
  \xdef\WF@ps{\WF@ps\WF@wrapil}\advance\@tempcnta\@ne
 }\endgroup
 \ifx\WF@pspars\@empty
  \global\@@parshape\c@WF@wrappedlines \WF@ps \WF@noil
 \else
  \WF@modps
\fi\fi}
\def\WF@modps{\begingroup
  \afterassignment\@tempdimb \@tempdima\WF@pspars
  \advance\@tempdima-\WF@adjtlm \advance\@tempdimb\WF@adjlw
  \let\WF@wrapil\WF@pspars
  \edef\@tempb{\global\@@parshape\c@WF@wrappedlines \WF@ps
     \the\@tempdima \the\@tempdimb}%
  \expandafter\endgroup\@tempb}
\def\WF@finale{\ifx\parshape\WF@fudgeparshape
 \WF@restoretol \let\@setpar\@@setpar \let\par\WF@@par
 \advance\linewidth\WF@adjlw \advance\@totalleftmargin-\WF@adjtlm
 \WF@info{Finish wrapping text}%
 \ifx\par\@@par \def\@par{\let\par\@@par\par}\else \let\@par\WF@@par \fi
 \let\parshape\@@parshape
 \global\parshape\ifx\WF@pspars\@empty \z@ \else \@ne \WF@pspars\fi \fi
 \ifvoid\WF@box \ifx\everypar\WF@everypar
  \let\everypar\WF@@everypar \everypar\expandafter{\the\WF@everypar}%
 \fi\fi}

% Full-page chapter plate: a themed frontispiece that faces the opening of
% each chapter. The art is laid in edge to edge as a single piece with NO
% margins, rules or frame: the page IS the artwork. Placement is relative to
% the paper corner rather than the text block, so the margin settings never
% intrude. Carries no folio, head or caption, and is skipped by the
% list-of-figures machinery on purpose.
%
% Each plate file is pre-cropped to the exact 8.5 x 11 page shape at 300 dpi,
% so the graphic is placed at its true size and nothing is stretched.
%
% The plate counts as the FIRST page of the chapter it introduces, so the
% contents list quotes the art page rather than the page the prose starts on.
% \chapterplate records its own page number and a link anchor; the next
% chapter-level contents entry is then written with those values in place of
% the ones LaTeX would have supplied. In the digital edition this also makes
% the contents link land on the art.

% must be genuinely blank -- book's default leaves the running head and folio
% standing on an otherwise empty leaf, which reads like a printing fault when it

% ---- dynamic figure boxes ------------------------------------------------
% A wrapped illustration is a fixed-width box of variable height: the narrower
% the column, the more lines the caption and credit take, and a long caption in
% a 0.28\textwidth box can end up taller than half the page. Such a box cannot
% be wrapped by any real paragraph, so it floats, lands alone at the top of the
% next page and leaves a hole.
%
% The cure is to let the box choose its own width. \ufo@grow measures the box,
% and while it is taller than the ceiling it widens it in 4%-of-measure steps
% (which shortens the caption) up to a hard cap. The requested width is a
% MINIMUM, never reduced, so nothing already well set changes.
\newdimen\ufo@wd
\newcount\ufo@gcnt
\newcommand*{\ufo@wfrac}{}

\newcommand{\ufo@grow}[2]{% #1 height ceiling, #2 width-fraction cap (in pt)
  \ufo@gcnt\z@
  \@whilenum\ufo@gcnt<12\do{%
    \advance\ufo@gcnt\@ne
    \ifdim\figwrapneed>#1\relax
      \ifdim\ufo@wd<#2\relax
        \advance\ufo@wd 0.04\p@
        \ufo@setwfrac
        \ufo@measurebox
      \fi
    \fi}}
% Hard height limiter. Widening shortens the caption but cannot help once the
% caption is already one line: past that point the picture itself is what makes
% the box tall. A wrapped box only works if the paragraph beside it is at least
% as deep as the box, and the biographical entries in this book run about ten
% lines. So clamp the box: shrink the image in 0.1in steps, down to a floor,
% until the box is short enough for a normal paragraph to sit beside it.
\newcommand{\ufo@shrinkto}[2]{% #1 height limit, #2 image-height floor
  \ufo@gcnt\z@

    \advance\ufo@gcnt\@ne
    \ifdim\figwrapneed>#1\relax
      \ifdim\ufo@ic>#2\relax
        \advance\ufo@ic-0.1in
        \ufo@measurebox
      \fi
    \fi}}
% Second half of the dynamic behaviour: fit the box to the room actually left
% on THIS page. Widening shortens the caption, but a box can still be too tall
% simply because it was called eight lines from the bottom. Rather than let it
% float away and leave the hole, shrink the picture -- in 0.15in steps, down to
% a floor of 1.35in, which is still a legible portrait -- until the whole box
% fits the space remaining. Only if even the floor will not fit does the figure
% float, which is now genuinely rare.

      \advance\ufo@gcnt\@ne

          \advance\ufo@ic-0.1in

        \fi
      \fi}%
  \fi}

% ---- wrapfig leak guard -------------------------------------------------
% wrapfig protects a wrapped figure with \parshape, not with real vertical
% space. If the paragraph beside the figure runs out before the figure does
% (short bio, then a box, a source list or a new subsection), the wrapping is
% still notionally in force. Start a second wrapped figure in that state and
% wrapfig reports a collision and takes its conflict branch, which leaves a
% one-line \parshape installed permanently -- every later paragraph in the
% book is then indented by the width of a figure that is no longer there.
%
% \ufowrapclear ends any wrapping still in force, but first reserves the
% vertical space the unfinished figure still needed, so nothing overlaps, and
% then clears \parshape by hand.
\makeatletter
% Two mechanisms are used, in order of reliability.
%
% (1) Geometric reservation. When a wrapped figure is set on the current page,
% \figwrap records the vertical position of the BOTTOM of the panel. Any block
% that cannot be wrapped then reserves exactly the distance still to go, and no
% more: if the text beside the picture already ran past it the deficit is zero
% and nothing is inserted, and if the text stopped short the block is pushed to
% just below the credit line. This is what makes overprinting impossible while
% still leaving no gratuitous whitespace.
%
% (2) Line-count fallback, used only when the panel was deferred as a float and
% its position on the page is not knowable here. The reservation is then capped
% at three lines, because reserving every remaining wrapped line verbatim is
% what produced the tall blank blocks between a short biography and the next
% heading.
